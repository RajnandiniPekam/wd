<div class="card">

<h3>1. Indexed Array</h3>

<?php

$tech_stack = array("MySQL", "HTML5", "CSS3", "MongoDB", "JavaScript");

$tech_stack[] = "Bootstrap 5";

echo "<ul>";

foreach($tech_stack as $tech){
    echo "<li>$tech</li>";
}

echo "</ul>";

echo "<b>Total Technologies: " . count($tech_stack) . "</b>";

?>

</div>

<!-- Associative Array -->
<div class="card">

<h3>2. Associative Array</h3>

<?php

$marks = array(
    "Operating Systems" => 88,
    "Digital Electronics" => 90,
    "Internet of Things" => 95,
    "Web Development" => 86
);

?>

<table>

<tr>
<th>Subject Name</th>
<th>Marks Obtained</th>
</tr>

<?php

foreach($marks as $subject => $score){

    echo "<tr>";
    echo "<td>$subject</td>";
    echo "<td>$score</td>";
    echo "</tr>";
}

?>

</table>
