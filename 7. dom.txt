    <button onclick="changeText()">Change Text</button>
    <button onclick="changeColor()">Change Color</button>
    <button onclick="addItem()">Add New Message</button>
    <button onclick="toggleBox()">Show / Hide</button>
    <button onclick="changeImage()">Change Image</button>

    <ul id="list"></ul>

<script>
    function changeText(){
        document.getElementById("message").innerHTML = "Text changed";
    }

    function changeColor(){
        document.getElementById("box").style.backgroundColor = "green";
    }

    function addItem(){
        var newItem = document.createElement("li");
        newItem.innerHTML = "Welcome to Development Era!";
        document.getElementById("list").appendChild(newItem);
    }

    function toggleBox(){
        var box = document.getElementById("box");
        if(box.style.display === "none"){
            box.style.display = "block";
        }else{
            box.style.display = "none";
        }
    }

    function changeImage(){
        document.getElementById("myImage").src = "Cat3.jpg";
    }
</script>
