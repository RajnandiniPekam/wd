<label>Full Name:</label>

<input type="text"
       name="fname"
       placeholder="Enter your name"
       required>

<label>Age:</label>

<input type="number"
       name="age"
       placeholder="Enter your age"
       required>

<input type="submit"
       value="Check Eligibility">

</form>

<?php

if ($_SERVER["REQUEST_METHOD"] == "POST") {

    $name = htmlspecialchars($_POST['fname']);

    $age = $_POST['age'];

    if (empty($name) || empty($age)) {

        echo "<p class='error'>Fields cannot be empty!</p>";

    } else {

        if ($age >= 18) {

            $status = "You are <b>Eligible</b> to vote.";

        } else {

            $years_left = 18 - $age;

            $status = "You are <b>Not Eligible</b>. Wait for $years_left more years.";
        }

        echo "<div class='result-box'>";

        echo "<h4>Result for $name:</h4>";

        echo "Your Age: $age <br><br>";

        echo $status;

        echo "</div>";
    }
}

?>
